export function buildImagePrompt(item: { title: string; price: number; bullets: string[]; format?: "square" | "horizontal" }): string {
  const aspect = item.format === "horizontal" ? "1200x630 landscape" : "1080x1080 square";
  const bulletsText = item.bullets.map(b => `"${b}"`).join(", ");
  return `Generate a premium product card for a Russian rental marketplace. 
    Aspect ratio: ${aspect}. Center the product from the attached photo, remove original background, replace with a stylish studio or atmospheric setting.
    Overlay the following Russian text clearly on the image:
    - Title: "${item.title}"
    - Price: "Аренда ${item.price} ₽/сутки"
    - Bullets: ${bulletsText}
    Style: minimalistic, modern, clean typography, no distortion, text must be perfectly readable.`;
}
